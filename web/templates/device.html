Install & renew automatically (Linux / macOS)

The URLs below contain a secret token that lets the device download its own certificate and key without logging in. Treat it like a password.

sudo mkdir -p /etc/privnet-tls/{{.Name}} && cd /etc/privnet-tls/{{.Name}}
sudo curl -fsS -o fullchain.pem {{.FetchBase}}/fullchain.pem
sudo curl -fsS -o privkey.pem {{.FetchBase}}/privkey.pem
sudo chmod 600 privkey.pem

Point your web server at the files, for example nginx:

server {
    listen 443 ssl;
    server_name {{.FQDN}};
    ssl_certificate     /etc/privnet-tls/{{.Name}}/fullchain.pem;
    ssl_certificate_key /etc/privnet-tls/{{.Name}}/privkey.pem;
    location / { proxy_pass http://127.0.0.1:8080; }
}

Refresh daily and reload when the certificate has changed (adjust the reload command):

sudo tee /etc/cron.daily/privnet-tls-{{.Name}} >/dev/null <<'EOF'
#!/bin/sh
set -e
cd /etc/privnet-tls/{{.Name}}
curl -fsS -o fullchain.pem.new {{.FetchBase}}/fullchain.pem
curl -fsS -o privkey.pem.new {{.FetchBase}}/privkey.pem
if ! cmp -s fullchain.pem.new fullchain.pem; then
  chmod 600 privkey.pem.new
  mv privkey.pem.new privkey.pem
  mv fullchain.pem.new fullchain.pem
  systemctl reload nginx
else
  rm -f fullchain.pem.new privkey.pem.new
fi
EOF
sudo chmod +x /etc/cron.daily/privnet-tls-{{.Name}}
